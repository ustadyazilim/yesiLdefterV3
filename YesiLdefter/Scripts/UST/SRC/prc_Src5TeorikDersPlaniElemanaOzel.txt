/*CreateProcedure*/

CREATE PROCEDURE [dbo].[prc_Src5TeorikDersPlaniElemanaOzel] (
   @FirmId int 
  ,@DonemTipiId int
  ,@GrupTipiId int 
  ,@DerslikAdiTipiId int 
  ,@PersonelId varchar(30)
  ,@ElemanGrubu varchar(30))  
AS BEGIN

-- DonemTipiId             : Dönem
-- GrupTipiId, SubeTipiId  : Gruplar
-- DerslikAdiTipiId        : Derslik
-- PersonelId              : Usta Öğretici

 
 --testlerin başı 
 /* 
 declare @FirmId int = 4204
 declare @DonemTipiId Int 
 declare @GrupTipiId Int 
 declare @DerslikAdiTipiId Int 
 declare @PersonelId varchar(30)
 declare @ElemanGrubu varchar(30)
 
 set @ElemanGrubu = 'Dönem'
 set @ElemanGrubu = 'Usta Eğitici'--'Gruplar'

 set @DonemTipiId = 202601
 set @PersonelId = '12345678901'
 
 set @GrupTipiId = 1
 -- set @ElemanGrubu = 'Derslikler'
 -- testlerin sonu
 */
 
 if (@DonemTipiId <= 0)
 begin
     set @DonemTipiId = dbo.fnc_getDonemTipiId ('BUGÜN', null)
 end

 declare @EgitimBaslangicTarihi date
 declare @EgitimBaslangicTarihi_ varchar(100)
 declare @EgitimBitisTarihi date
  
 
 Select @EgitimBaslangicTarihi = isnull(min(GrupBaslamaTarihi), getdate())
 From  dbo.Src5Donemi
 Where FirmId = @FirmId
 and   DonemTipiId = @DonemTipiId

 set @EgitimBaslangicTarihi_ = 'Convert(datetime, ''' + CONVERT(varchar(10), @EgitimBaslangicTarihi, 103) + ''', 103) '

 print @EgitimBaslangicTarihi_

 --print @EgitimBaslangicTarihi
 --print @EgitimBitisTarihi

 declare @sql nvarchar(max)
 declare @sqlWhere nvarchar(500)
 declare @sqlOrderBy nvarchar(500)

 set @sql = '
   Select 
   td.[Id]
 , td.[FirmId]
 , td.[IsActive]
 , td.[DonemTipiId]
 , td.[GrupTipiId]

 , td.[DersTipiId]
 , td.[DersSiraNo]
 , td.[DerslikAdiTipiId]
 , td.[DersTarihi]
 , td.[DersSaatiTipiId]
 , td.[AsilPersonelId]
 , td.[YedekPersonelId]

 , td.[DersSuresiDakika]
 , td.[BaslamaSaati]
 , td.[BitisSaati]
 , td.[KamuBaslangicSaati]
 , td.[KamuBitisSaati]

 , td.[TDersIsUploaded]
 , td.[OzelGunlerId]
 , td.[DersHaftasiId]

 -- Join Tables standart fields 
 , [personel].TamAdiSoyadi as Lkp_PersonelTamAdiSoyadi
 , [personel].TcVkNo as Lkp_PersonelTcNo
 , [personel].TamAdiSoyadi + CHAR(13) + CHAR(10) + FNo23.DerslikAdiTipi as Lkp_DerslikVePersonel
 , [personel].TamAdiSoyadi + '', '' + FNo23.DerslikAdiTipi + '', '' + FNo11.DonemTipi + '', '' + FNo12.GrupTipi as Lkp_DerslikPersDonemGrup
 , [dersler].DersAdi + ''  '' + convert(varchar(2),td.DersSiraNo) + ''.Drs'' + CHAR(13) + CHAR(10) + personel.TamAdiSoyadi + CHAR(13) + CHAR(10) + FNo23.DerslikAdiTipi as Lkp_DersTipi
 , ( case 
     when td.DersTipiId >= 2010 then convert(int,dersler.MebbisKodu) - 85 
     end ) as Lkp_LabelId  
 
 -- LookUp Fields
 , FNo11.DonemTipi as Lkp_DonemTipi
 , FNo12.GrupTipi as Lkp_GrupTipi
 , FNo21.DersTipi as Lkp_DersTipi
 , FNo23.DerslikAdiTipi as Lkp_DerslikAdiTipi
 -- Özel field : Donem + Grup + Sube 
 , FNo11.DonemTipi + '' , '' + FNo12.GrupTipi as Lkp_DonemGrup
 
 from dbo.Src5TeorikDers as td with (nolock) 

   -- Join Tables 
   left outer join dbo.SrcPersoneli as [personel] with (nolock)   on ( td.AsilPersonelId = [personel].TcVkNo  ) 
   left outer join Lkp.SrcDersler   as [dersler] with (nolock) on ( td.DersTipiId = [dersler].Id  ) 
   -- LookUp Tables
   left outer join Lkp.SrcDonemTipi as FNo11 with (nolock) on ( td.DonemTipiId = FNo11.Id ) 
   left outer join Lkp.Src5GrupTipi as FNo12 with (nolock) on ( td.GrupTipiId = FNo12.Id ) 
   left outer join Lkp.Src5TeorikDersDersTipi as FNo21 with (nolock) on ( td.DersTipiId = FNo21.Id ) 
   left outer join Lkp.Src5TeorikDersDerslikAdiTipi as FNo23 with (nolock) on ( td.DerslikAdiTipiId = FNo23.Id ) 

 Where td.FirmId = ' + convert(varchar(10),@FirmId) + ' '

 set @sqlOrderBy = '
   or (    td.OzelGunlerId > 0 
       and td.BaslamaSaati > DATEADD( mm, -1, ' + @EgitimBaslangicTarihi_ + ') 
       and td.BitisSaati   < DATEADD( mm,  3, ' + @EgitimBaslangicTarihi_ + ') 
      )
   Order by td.BaslamaSaati '

 -- Dönem ve grupla bilgisi isteniyorsa DonemTipiId koşulu olsun
 --if (@ElemanGrubu = 'Dönem' or @ElemanGrubu = 'Gruplar') -- bunu açınca ilk okumada boş data geliyor ve dataset formun üzerine işlenmiyor
 set @sqlWhere = '  and td.DonemTipiId = ' + Convert(varchar(30), @DonemTipiId) + ' '

 -- Elemenanlar ver derslikler onlunca grubun başlangıç tarihinden sonra olan tüm diğer gruplardaki derslikleride getirsin
 if (@ElemanGrubu = 'Derslikler' or @ElemanGrubu = 'Usta Eğitici')
     set @sqlWhere = '  and Convert(date, td.DersTarihi, 103 ) >= Convert(date, ''' + Convert(varchar(30), Format(@EgitimBaslangicTarihi, 'dd.MM.yyyy')) + ''', 103) '

 --set @ElemanGrubu = 'Dönem'
 --set @ElemanGrubu = 'Gruplar'
 --set @ElemanGrubu = 'Derslikler'
 --set @ElemanGrubu = 'Usta Eğitici'

 --set @GrupTipiId = 1
 --set @DerslikAdiTipiId = 1
 --set @PersonelId = '205013645624899268'
 
 if ((@GrupTipiId > 0) and (@ElemanGrubu = 'Gruplar'))
   set @sqlWhere = @sqlWhere +
     '  and td.GrupTipiId = ' + Convert(varchar(20), @GrupTipiId) + ' '

 if ((@DerslikAdiTipiId > 0) and (@ElemanGrubu = 'Derslikler'))
   set @sqlWhere = @sqlWhere +
   '  and td.DerslikAdiTipiId = ' + Convert(varchar(20), @DerslikAdiTipiId) + ' '

 if ((@PersonelId <> '') and (@ElemanGrubu = 'Usta Eğitici')) 
   set @sqlWhere = @sqlWhere +
   '  and td.AsilPersonelId = ''' + @PersonelId + ''''

 set @sql = @sql + @sqlWhere + @sqlOrderBy

 --print @sql
 EXEC sp_executesql @sql
  
END -- procedure
