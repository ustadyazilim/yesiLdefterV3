/*CreateTrigger*/

CREATE TRIGGER [dbo].[trg_Src5Donemi] on [dbo].[Src5Donemi]
AFTER INSERT, UPDATE
AS BEGIN

    DECLARE cursorSrc5Donemi cursor local for select Id from Inserted
    DECLARE @curId bigint

    OPEN cursorSrc5Donemi

    FETCH NEXT FROM cursorSrc5Donemi INTO @curId

    declare @FirmId Int = 0
    declare @DonemTipiId int = 0
    declare @GrupTipiId int = 0
    declare @SertifikaTipiId smallint = 0
    declare @SertifikaSinifTipiId smallInt = 0
    declare @GrupSiraNo smallInt = 0
    declare @DonemAdi nvarchar(100)

    declare @DonemYilAy varchar(10)
    declare @GrupTipi varchar(4)
    declare @SertifikaSinifAdi varchar(50)
    declare @GrupSiraNoAdi varchar(10)

    WHILE @@FETCH_STATUS = 0
    BEGIN

       Select @FirmId = ins.FirmId
         , @DonemTipiId = DonemTipiId
         , @GrupTipiId  = GrupTipiId
         , @SertifikaTipiId = SertifikaTipiId
         , @SertifikaSinifTipiId = SertifikaSinifTipiId
         , @GrupSiraNo = GrupSiraNo

       From inserted as ins
       Where Id = @curId

       Select @SertifikaSinifAdi = SertifikaSinifTipi
       From  Lkp.Src5DonemiSertifikaSinifTipi
       Where Id = @SertifikaSinifTipiId

       set @DonemYilAy = substring(convert(varchar(10), @DonemTipiId),1,4) + '.' + substring(convert(varchar(10), @DonemTipiId),5,2)
       set @GrupTipi = 'G' + RIGHT(REPLICATE('0', 2) + CAST(@GrupTipiId AS VARCHAR(5)), 2) 
       
       set @GrupSiraNoAdi = '';
       if (@GrupSiraNo > 0)
           set @GrupSiraNoAdi = '  [' + RIGHT(REPLICATE('0', 4) + CAST(@GrupSiraNo AS VARCHAR(5)), 4) + ']'

       Set @DonemAdi = @DonemYilAy +'/'+ @GrupTipi + ': ' + @SertifikaSinifAdi + @GrupSiraNoAdi
       -- select @DonemAdi

       Update dbo.Src5Donemi set DonemAdi = @DonemAdi
       Where Id = @curId

       
       FETCH NEXT FROM cursorSrc5Donemi INTO @curId
    END -- While

    CLOSE cursorSrc5Donemi
    DEALLOCATE cursorSrc5Donemi   


END -- create trigger

/*CreateTriggerEnd*/


/*CreateTrigger*/

CREATE TRIGGER [dbo].[trg_Src5DonemiDeleteIsLock] on [dbo].[Src5Donemi]
INSTEAD of DELETE
AS BEGIN

    declare isLockCursorDlt cursor local for select Id from deleted
    declare @curId bigint

    OPEN isLockCursorDlt

    FETCH NEXT FROM isLockCursorDlt INTO @curId
	   
    WHILE @@FETCH_STATUS = 0
    BEGIN
        if ( Select COUNT(t.Id) as Adet from deleted as d
             left join dbo.SrcAdayTalep as t on ( d.DonemTipiId = t.DonemTipiId and d.GrupTipiId = t.GrupTipiId and t.TalepTipiId = 2 ) 
           ) > 0 
        begin
            PRINT 'Bu kayıt silinemez!'
            RETURN
        end
	    IF EXISTS (SELECT 1 FROM deleted WHERE isnull(GrupTipiId,0) = 1 and Id = @curId)
        BEGIN
            PRINT 'Bu kayıt silinemez!'
            RETURN
        END

        Delete from dbo.Src5Donemi  Where Id = @curId and isnull(GrupTipiId,0) <> 1

      FETCH NEXT FROM isLockCursorDlt INTO @curId
    END

    CLOSE isLockCursorDlt
    DEALLOCATE isLockCursorDlt

END

 /*CreateTriggerEnd*/
