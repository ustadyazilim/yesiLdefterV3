                 ,SubeTipiId = @ySubeTipiId
                 ,GrupSubeId = @kontrol1GrupSubeId
              Where Id = @TalepId 

              Set @eGrupSubeId = @kontrol1GrupSubeId
              Set @yGrupSubeId = @kontrol1GrupSubeId
          end

          -- GrupSubeId değiştiyse
          if (@eDonemTipiId = @yDonemTipiId
          and @eGrupTipiId = @yGrupTipiId
          and @eSubeTipiId = @ySubeTipiId
          ) and (@eGrupSubeId <> @yGrupSubeId)
          begin
              Select 
                  @kontrol2DonemTipiId = isnull(DonemTipiId,0)
                 ,@kontrol2GrupTipiId = isnull(GrupTipiId,0)
                 ,@kontrol2SubeTipiId = isnull(SubeTipiId,0)
              From dbo.SrcGrupSubesi
              Where FirmId = @yFirmId 
              and   Id = @yGrupSubeId

              --Select @kontrol2DonemTipiId,@kontrol2GrupTipiId,@kontrol2SubeTipiId,'yy'
              Update dbo.SrcAdayTalep Set
                  DonemTipiId = @kontrol2DonemTipiId
                 ,GrupTipiId = @kontrol2GrupTipiId
                 ,SubeTipiId = @kontrol2SubeTipiId
              Where Id = @TalepId 
          end 

          -- Donem veya Grup veya Şube ve hatta GrupSubeId değiştiyse
          -- GrupSubeId baz alınıyor
          if (@eDonemTipiId <> @yDonemTipiId
          or @eGrupTipiId <> @yGrupTipiId
          or @eSubeTipiId <> @ySubeTipiId
          ) and (@eGrupSubeId <> @yGrupSubeId)
          begin
              Select 
                  @kontrol2DonemTipiId = isnull(DonemTipiId,0)
                 ,@kontrol2GrupTipiId = isnull(GrupTipiId,0)
                 ,@kontrol2SubeTipiId = isnull(SubeTipiId,0)
              From dbo.SrcGrupSubesi
              Where FirmId = @yFirmId 
              and   Id = @yGrupSubeId

              --Select @kontrol2DonemTipiId,@kontrol2GrupTipiId,@kontrol2SubeTipiId,'xx'
              Update dbo.SrcAdayTalep Set
                  DonemTipiId = @kontrol2DonemTipiId
                 ,GrupTipiId = @kontrol2GrupTipiId
                 ,SubeTipiId = @kontrol2SubeTipiId
              Where Id = @TalepId 
          end 

          -- Eğer bu belge (Artı/2. Belge) ise
          if (@yArtiBelgeTipiId = 2)
          begin
              -- 1. belge sertifika almadığı müddetce dönem belirlenemez
              if ( Select KayitTipiId From dbo.SrcAdayTakip --Talep
                   Where TalepId = @yParentId ) < 4 
              and ( @yDonemTipiId <> 0 or @yGrupTipiId <> 0 or @ySubeTipiId <> 0)
              begin
                  Update dbo.SrcAdayTalep Set
                      DonemTipiId = 0
                    , GrupTipiId = 1
                    , SubeTipiId = 0
                    , GrupSubeId = 0
                  Where Id = @TalepId
              end
          end -- if 
         
          if (@yIsDosyaKapandi = 1 and @yKayitTipiId <> 8)
          begin 
              Update SrcAdayTalep Set KayitTipiId = 8
                , DosyaKapanmaTarihi = GETDATE()
              Where Id = @TalepId
          end

          if (@yIsDosyaKapandi = 0 and @yKayitTipiId = 8)
          begin 
              Update SrcAdayTalep Set KayitTipiId = 3
                , DosyaKapanmaTarihi = GETDATE()
              Where Id = @TalepId
          end

	      --Execute [dbo].[prc_SrcAdayTalepYasiniHesapla] @TalepId
          
          -- Veri transferi yapılmıyorsa 
          if (@DbUpdatesIsActive = 0)
          begin
              Execute [dbo].[prc_SrcAdayTablesInsert] @TalepId  -- Data transferi sırasında oluşturuluyor zaten insert oluyor
              Execute [dbo].[prc_SrcAdayTakipTalepKaydiSingleSet] @TalepId
              Execute [dbo].[prc_SrcAdayTakip] @yFirmId, @TalepId, 0, 'SrcAdayTakipKayit'
          end
		  
          -- SrcAdayBelgeler tablosu
		  Update SrcAdayBelgeler Set DonemTipiId = @yDonemTipiId
		  Where TalepId = @TalepId
		  and isnull(DonemTipiId,0) <> @yDonemTipiId
      end 	  

      FETCH NEXT FROM talepCursor_ INTO @TalepId
      print '  talep while end'
  END -- While

  CLOSE talepCursor_
  DEALLOCATE talepCursor_   

  print 'talep trigger End'

END -- create trigger

/*CreateTriggerEnd*/

CREATE TRIGGER [dbo].[trg_SrcAdayTalepDelete] on [dbo].[SrcAdayTalep]
AFTER DELETE
AS BEGIN

    declare talepCursorDlt cursor local for select Id from deleted
	declare @curId bigint

	OPEN talepCursorDlt

    FETCH NEXT FROM talepCursorDlt INTO @curId

    WHILE @@FETCH_STATUS = 0
    BEGIN
	  Delete from SrcAdayAdliSicil  Where TalepId = @curId
	  Delete from SrcAdayAdres      Where TalepId = @curId
	  Delete from SrcAdayBelgeler   Where TalepId = @curId
	  Delete from SrcAdayOgrenim    Where TalepId = @curId
	  Delete from SrcAdayTakip      Where TalepId = @curId
	  Delete from SrcAdayUcret      Where TalepId = @curId

      FETCH NEXT FROM talepCursorDlt INTO @curId
    END

    CLOSE talepCursorDlt
    DEALLOCATE talepCursorDlt

END
