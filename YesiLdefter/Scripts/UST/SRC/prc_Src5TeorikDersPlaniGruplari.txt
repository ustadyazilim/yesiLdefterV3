CREATE PROCEDURE [dbo].[prc_Src5TeorikDersPlaniGruplari] (@FirmId int, @DonemTipiId int)  
AS BEGIN

-- DonemTipiId        : Dönem
-- GrupTipiId         : Gruplar
-- PersonelId         : Usta Öğretici
-- DerslikAdiTipiId   : Derslikler

   --declare @DonemTipiId Int 
   --set @DonemTipiId = 202601

   Select  
     @DonemTipiId as DonemTipiId
   , 0  as GrupTipiId
   , 0  as DerslikAdiTipiId 
   , '' as AsilPersonelId
   
   , 'Dönem' as Lkp_ElemanGrubu
   , donem.DonemTipi as Lkp_ElemanAdi
   
   from Lkp.SrcDonemTipi as donem
   Where donem.Id = @DonemTipiId

   union all

   Select  
     @DonemTipiId as DonemTipiId
   , x.GrupTipiId as GrupTipiId
   , 0  as DerslikAdiTipiId 
   , '' as AsilPersonelId
   
   , 'Gruplar' as Lkp_ElemanGrubu
   , x.GrupTipi as Lkp_ElemanAdi
   
   from (
     Select distinct [teoDers].GrupTipiId, grup.GrupTipi
     from Src5TeorikDers as [teoDers]
	   left outer join Lkp.Src5GrupTipi as grup on ( teoDers.GrupTipiId = grup.Id )
     Where [teoDers].DonemTipiId = @DonemTipiId
     and   [teoDers].FirmId = @FirmId
   ) as x

   union all

   Select  
     @DonemTipiId as DonemTipiId
   , 0  as GrupTipiId
   , 0  as DerslikAdiTipiId 
   , x.AsilPersonelId
   
   , 'Usta Eğitici' as Lkp_ElemanGrubu
   , x.TamAdiSoyadi as Lkp_ElemanAdi
   
   from (
     Select distinct [teoDers].AsilPersonelId, per.TamAdiSoyadi
     from Src5TeorikDers as [teoDers]
	   left outer join SrcPersoneli as [per] on (teoDers.AsilPersonelId = per.TcvKNo)
     Where [teoDers].DonemTipiId = @DonemTipiId
     and   [teoDers].FirmId = @FirmId
   ) x

   union all

   Select  
     @DonemTipiId as DonemTipiId
   , 0  as GrupTipiId
   , x.DerslikAdiTipiId 
   , '' as AsilPersonelId
   
   , 'Derslikler' as Lkp_ElemanGrubu
   , x.DerslikAdiTipi as Lkp_ElemanAdi
   
   from (
     Select distinct [teoDers].DerslikAdiTipiId, derslik.DerslikAdiTipi
     from Src5TeorikDers as [teoDers]
	   left outer join Lkp.Src5TeorikDersDerslikAdiTipi as [derslik] on (teoDers.DerslikAdiTipiId = derslik.Id)
     Where [teoDers].DonemTipiId = @DonemTipiId
     and   [teoDers].FirmId = @FirmId
   ) x

END -- procedure
