CREATE TRIGGER [dbo].[trg_Src5TeorikDers] on [dbo].[Src5TeorikDers] 
AFTER INSERT, UPDATE
AS BEGIN

    DECLARE myCursor cursor local for select Id from Inserted
    DECLARE @curId bigint

    OPEN myCursor
    FETCH NEXT FROM myCursor INTO @curId

    declare @eDersTarihi date = null
    declare @eDersSaatiTipiId smallInt = -1
	declare @eBaslamaSaati datetime
	declare @eBitisSaati datetime

    declare @yFirmId int = 0
    declare @yDonemTipiId int = 0
    declare @yGrupTipiId smallInt = 0
    
    declare @yDersTarihi date = null
    declare @yDersSaatiTipiId smallInt = -1
    declare @yPersonelId varchar(20) = ''
    declare @yPersonelYedekId varchar(20) = ''

	declare @yDersSuresiDakika smallint
    declare @yBaslamaSaati datetime
	declare @yBitisSaati datetime
    declare @CurrentEnd datetime
    declare @yBaslamaSaatiChechked varchar(30)
    declare @yBitisSaatiChechked varchar(30)
    
    declare @TcNo varchar(20) = null
    declare @TDersIsUploaded bit = 0

	declare @insId int = 0
    declare @mvcId int = 0

    WHILE @@FETCH_STATUS = 0
    BEGIN
    	
      Select
          @eDersTarihi = dlt.DersTarihi
         ,@eDersSaatiTipiId = dlt.DersSaatiTipiId
         ,@eBaslamaSaati = dlt.BaslamaSaati
         ,@eBitisSaati = dlt.BitisSaati	  
      From deleted dlt
      Where dlt.Id = @curId

      Select
	      @yFirmId = ins.FirmId
         ,@yDonemTipiId = ins.DonemTipiId
         ,@yGrupTipiId = ins.GrupTipiId
      
         ,@yDersTarihi = ins.DersTarihi
         ,@yDersSaatiTipiId = ins.DersSaatiTipiId
         ,@yPersonelId = isnull(ins.AsilPersonelId,'')
         ,@yPersonelYedekId = isnull(ins.YedekPersonelId,'')
         ,@TDersIsUploaded = isnull(ins.TDersIsUploaded,0)
         ,@yDersSuresiDakika = isnull(ins.DersSuresiDakika,50)
         ,@yBaslamaSaati = ins.BaslamaSaati
         ,@yBitisSaati = ins.BitisSaati	  
         ,@yBaslamaSaatiChechked = Convert(varchar(30), ins.BaslamaSaati, 121) -- saati al
         ,@yBitisSaatiChechked = Convert(varchar(30), ins.BitisSaati, 121) -- saati al
      From inserted ins
      Where ins.Id = @curId


      -- personelId kontrolu
      -- Kamudan okuma yapınca alınan ismi alınıyor
      -- alınan bu ismi TcNoIzinNo ya çeviriyor
      if (@yPersonelId != '')
      begin 
          Set @TcNo = ''
          Select @TcNo = isnull(TcVkNo,'')
          From dbo.SrcPersoneli
          Where TamAdiSoyadi like @yPersonelId+'%'
      
          if (@TcNo <> '')
          begin
              update dbo.Src5TeorikDers set AsilPersonelId = @TcNo
              Where Id = @curId
          end
      end

      if (@yPersonelYedekId != '')
      begin 
          Set @TcNo = ''
          Select @TcNo = isnull(TcVkNo,'')
          From dbo.SrcPersoneli
          Where TamAdiSoyadi like @yPersonelYedekId+'%'
            
          if (@TcNo <> '')
          begin
              update dbo.Src5TeorikDers set YedekPersonelId = @TcNo
              Where Id = @curId
          end
      end
      
      if (@TDersIsUploaded = 1) -- yani Mebbise yüklenmişse, değşişklik olamaz
      begin
          update dbo.Src5TeorikDers set 
			    DersTarihi = @eDersTarihi
              , BaslamaSaati = @eBaslamaSaati
              , BitisSaati   = @eBitisSaati
          Where Id = @curId
          return
      end

      if (convert(date,@yDersTarihi,103) <> convert(date,@yBaslamaSaati,103))
      begin
          if (convert(date, @eDersTarihi,103) <> convert(date,@yDersTarihi,103))
          begin
             print 'tarih değişti' 
             declare @BasSaat varchar(5) = convert(varchar(5), convert(time,@yBaslamaSaati))
             Set @yBaslamaSaati = CAST(@yDersTarihi AS DATETIME) + CAST(@BasSaat AS DATETIME);
          end
          else begin
             print 'tarih ve saat değişti'
             Set @yDersTarihi = convert(date,@yBaslamaSaati,103)
          end
      end

      -- DersTarihi değiştiyse
      if (@eDersTarihi <> @yDersTarihi)
      begin
         Update dbo.Src5TeorikDers Set
             DersTarihi = @yDersTarihi
         Where Id = @curId
      end
      -- Başlangıç saati değiştiyse
      if (@eBaslamaSaati <> @yBaslamaSaati)
      begin
         Update dbo.Src5TeorikDers Set
             BaslamaSaati = @yBaslamaSaati
         Where Id = @curId
      end
      -- Bitiş saati değiştiyse
      Set @CurrentEnd = DATEADD(MINUTE, @yDersSuresiDakika, @yBaslamaSaati);
      if (@CurrentEnd <> @yBitisSaati)
      begin
         Set @yBitisSaati = @CurrentEnd
         Update dbo.Src5TeorikDers Set
             BitisSaati = @yBitisSaati
         Where Id = @curId
      end

      Update dbo.Src5TeorikDers Set
           KamuBaslangicSaati = substring(Convert(varchar(30), @yBaslamaSaati, 121),12,5) -- saati al
         , KamuBitisSaati = substring(Convert(varchar(30), @yBitisSaati, 121),12,5) -- saati al
         , DersHaftasiId = DatePart(ww, @yBaslamaSaati)
      Where Id = @curId

      --Execute [dbo].[prc_SrcTeorikDersIhlalleriSaatler] @eFirmId, @eDonemTipiId
      --Execute [dbo].[prc_SrcTeorikDersIhlalleriCakismalar] @eFirmId, @eDonemTipiId
      --Execute [dbo].[prc_SrcDersIhlalleriPersonel] @curId, @eFirmId, @eDonemTipiId, @ePersonelId

      FETCH NEXT FROM myCursor INTO @curId
    END

    --Execute [dbo].[prc_SrcGrupSubesiTeorikDersToplami] @eFirmId, @eDonemTipiId, @eGrupTipiId, @eSubeTipiId
    --Execute [dbo].[prc_SrcGrupSubesiTakipSet] @eFirmId, @eDonemTipiId, @eGrupTipiId, @eSubeTipiId

    CLOSE myCursor
    DEALLOCATE myCursor   


END -- create trigger

/*CreateTriggerEnd*/

CREATE TRIGGER [dbo].[trg_Src5TeorikDersDelete] on [dbo].[Src5TeorikDers]
AFTER DELETE
AS BEGIN

    declare teoCursorDlt cursor local for select Id from deleted
	declare @curId bigint

	OPEN teoCursorDlt
    FETCH NEXT FROM teoCursorDlt INTO @curId

    WHILE @@FETCH_STATUS = 0
    BEGIN

	    Delete From dbo.SrcKIhlalTeorikD 
		Where InsertedId = @curId 
		or    MevcutId   = @curId

      FETCH NEXT FROM teoCursorDlt INTO @curId
    END

    CLOSE teoCursorDlt
    DEALLOCATE teoCursorDlt

END

    declare isLockCursorDlt cursor local for select Id from deleted

	declare @PersonelId varchar(30)

    OPEN isLockCursorDlt

    END
